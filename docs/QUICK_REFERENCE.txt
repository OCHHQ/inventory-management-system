#                sheet_name       item_name  qty unit price reorder

# UPDATE QUANTITY
manager.update_quantity('Cashier Items', 'Marker', 30)
#                        sheet_name       item_name  new_qty

# UPDATE PRICE
manager.update_price('Cleaning Items', 'Vim', 500)
#                     sheet_name        item_name  new_price

# GET LOW STOCK ITEMS
reorder_df = manager.get_items_to_reorder()
print(reorder_df)

# GENERATE REPORT
manager.generate_report('my_report.txt')

# EXPORT TO CSV
manager.export_to_csv('exports')

# SAVE CHANGES (IMPORTANT!)
manager.save()

import pandas as pd

df = pd.read_excel('inventory_tracker.xlsx', sheet_name='Cashier Items', skiprows=4)

for _, row in df.iterrows():
    if pd.notna(row['Unit Price (₦)']):
        old_price = row['Unit Price (₦)']
        new_price = old_price * 1.20
        manager.update_price('Cashier Items', row['Item Name'], new_price)

manager.save()

from datetime import datetime

manager = InventoryManager('inventory_tracker.xlsx')
reorder = manager.get_items_to_reorder()

if not reorder.empty:
    print("⚠️ LOW STOCK ALERT!")

    manager.generate_report(f'alert_{datetime.now().strftime("%Y%m%d")}.txt')

└──────────────────────────────────────────────────────────────────────────────┘

PROBLEM: "ModuleNotFoundError: No module named 'openpyxl'"
SOLUTION: pip install openpyxl pandas --break-system-packages

PROBLEM: Excel shows #DIV/0! error
SOLUTION: Some unit prices are 0. Enter a price for those items.

PROBLEM: "Item not found"
SOLUTION: Check spelling - item names are case-sensitive.

PROBLEM: Changes not saving
SOLUTION: Always call manager.save() after making changes!

PROBLEM: Excel file won't open
SOLUTION: Make sure Excel file is closed before running Python scripts.
